          package com.eddefi.calc

          import java.math.BigDecimal
          import java.math.MathContext

          data class CalcState(
              val expr: String = "",
              val lastExpr: String = "",
              val done: Boolean = false
          )

          fun isOp(c: Char): Boolean = c == '+' || c == '−' || c == '×' || c == '÷'

          private fun isUnaryAt(e: String, i: Int): Boolean = i == 0 || isOp(e[i - 1])

          private fun numStart(e: String): Int {
              var i = e.length
              while (i > 0 && (e[i - 1].isDigit() || e[i - 1] == '.')) i--
              return i
          }

          fun hasOperator(e: String): Boolean = e.drop(1).any { isOp(it) }

          object Engine {

              fun press(s: CalcState, key: String): CalcState = when (key) {
                  "C" -> CalcState()
                  "⌫" -> back(s)
                  "±" -> negate(s)
                  "%" -> percent(s)
                  "=" -> evaluateKey(s)
                  "+", "−", "×", "÷" -> op(s, key[0])
                  "." -> dot(s)
                  else -> digit(s, key[0])
              }

              private fun digit(s: CalcState, d: Char): CalcState {
                  val base = if (s.done) "" else s.expr
                  val start = numStart(base)
                  val cur = base.substring(start)
                  if (cur.count { it.isDigit() } >= 15) return s
                  val next = if (cur == "0") base.substring(0, start) + d else base + d
                  return CalcState(next)
              }

              private fun dot(s: CalcState): CalcState {
                  val base = if (s.done) "" else s.expr
                  val start = numStart(base)
                  val cur = base.substring(start)
                  if (cur.contains('.')) return s
                  val next = if (cur.isEmpty()) base + "0." else base + "."
                  return CalcState(next)
              }

              private fun op(s: CalcState, o: Char): CalcState {
                  var e = if (s.expr == "Error") "" else s.expr
                  if (e.isEmpty()) return if (o == '−') CalcState("−") else s
                  if (e == "−") return s
                  if (e.last() == '.') e = e.dropLast(1)
                  if (isOp(e.last())) {
                      val trimmed = e.dropLast(1)
                      if (trimmed.isEmpty() || isOp(trimmed.last())) return CalcState(e)
                      return CalcState(trimmed + o)
                  }
                  return CalcState(e + o)
              }

              private fun back(s: CalcState): CalcState {
                  if (s.done) return CalcState()
                  return CalcState(s.expr.dropLast(1))
              }

              private fun negate(s: CalcState): CalcState {
                  val e = if (s.expr == "Error") "" else s.expr
                  if (e.isEmpty()) return CalcState("−")
                  val start = numStart(e)
                  if (start == e.length) {
                      val last = e.last()
                      if (last == '−') {
                          return if (isUnaryAt(e, e.length - 1)) CalcState(e.dropLast(1)) else CalcState(e)
                      }
                      return CalcState(e + "−")
                  }
                  val m = start - 1
                  return if (m >= 0 && e[m] == '−' && isUnaryAt(e, m)) {
                      CalcState(e.removeRange(m, m + 1))
                  } else {
                      CalcState(e.substring(0, start) + "−" + e.substring(start))
                  }
              }

              private fun percent(s: CalcState): CalcState {
                  val e = s.expr
                  if (e.isEmpty() || e == "Error") return s
                  val start = numStart(e)
                  if (start == e.length) return s
                  val num = e.substring(start).toDoubleOrNull() ?: return s
                  return CalcState(e.substring(0, start) + fmt(num / 100.0))
              }

              private fun evaluateKey(s: CalcState): CalcState {
                  if (s.done || s.expr.isEmpty() || s.expr == "Error") return s
                  var e = s.expr
                  while (e.isNotEmpty() && (isOp(e.last()) || e.last() == '.')) e = e.dropLast(1)
                  if (e.isEmpty()) return s
                  val r = evaluate(e) ?: return CalcState("Error", e, true)
                  return CalcState(r, e, true)
              }
          }

          fun evaluate(e: String): String? {
              val nums = ArrayList<Double>()
              val ops = ArrayList<Char>()
              var i = 0
              while (i < e.length) {
                  var neg = false
                  if (e[i] == '−') {
                      neg = true
                      i++
                  }
                  val st = i
                  while (i < e.length && (e[i].isDigit() || e[i] == '.')) i++
                  if (st == i) return null
                  val n = e.substring(st, i).toDoubleOrNull() ?: return null
                  nums.add(if (neg) -n else n)
                  if (i < e.length) {
                      val o = e[i]
                      if (!isOp(o)) return null
                      ops.add(o)
                      i++
                  }
              }
              if (nums.size != ops.size + 1) return null

              val n2 = ArrayList<Double>()
              val o2 = ArrayList<Char>()
              n2.add(nums[0])
              for (k in ops.indices) {
                  val o = ops[k]
                  val v = nums[k + 1]
                  when (o) {
                      '×' -> n2[n2.size - 1] = n2[n2.size - 1] * v
                      '÷' -> {
                          if (v == 0.0) return null
                          n2[n2.size - 1] = n2[n2.size - 1] / v
                      }
                      else -> {
                          o2.add(o)
                          n2.add(v)
                      }
                  }
              }
              var acc = n2[0]
              for (k in o2.indices) {
                  acc = if (o2[k] == '+') acc + n2[k + 1] else acc - n2[k + 1]
              }
              if (acc.isNaN() || acc.isInfinite()) return null
              return fmt(acc)
          }

          fun fmt(d: Double): String {
              if (d == 0.0) return "0"
              val s = BigDecimal(d).round(MathContext(12)).stripTrailingZeros().toPlainString()
              return s.replace('-', '−')
          }

          fun group(s: String): String {
              var sign = ""
              var body = s
              if (body.startsWith("−")) {
                  sign = "−"
                  body = body.drop(1)
              }
              val parts = body.split('.', limit = 2)
              val ip = parts[0].reversed().chunked(3).joinToString(",").reversed()
              val frac = if (parts.size > 1) "." + parts[1] else ""
              return sign + ip + frac
          }

          fun pretty(e: String): String {
              val sb = StringBuilder()
              for (i in e.indices) {
                  val c = e[i]
                  val unary = c == '−' && (i == 0 || isOp(e[i - 1]))
                  if (isOp(c) && !unary) {
                      sb.append(' ').append(c).append(' ')
                  } else {
                      sb.append(c)
                  }
              }
              return sb.toString()
          }

          fun topText(s: CalcState): String {
              if (s.done) return if (s.lastExpr.isEmpty()) "" else pretty(s.lastExpr) + " ="
              return if (hasOperator(s.expr)) pretty(s.expr) else ""
          }

          fun bigText(s: CalcState): String {
              if (s.expr == "Error") return "Error"
              if (s.done) return group(s.expr)
              val e = s.expr
              if (e.isEmpty()) return "0"
              if (!hasOperator(e)) return group(e)
              var t = e
              while (t.isNotEmpty() && (isOp(t.last()) || t.last() == '.')) t = t.dropLast(1)
              if (t.isEmpty()) return "0"
              val r = evaluate(t) ?: return "Error"
              return group(r)
          }
